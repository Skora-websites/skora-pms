public function sendWhatsapp(Request $request)
{
    $validator = Validator::make($request->all(), [
        'mobile' => 'required|digits:10',
        'message' => 'required|string'
    ]);

    if ($validator->fails()) {
        return response()->json([
            'success' => false,
            'message' => $validator->errors()->first()
        ]);
    }

    $mobileNo = '+91' . $request->mobile;

    $formData = [
        ['name' => 'appkey', 'contents' => $this->appKey],
        ['name' => 'authkey', 'contents' => $this->authKey],
        ['name' => 'to', 'contents' => $mobileNo],
        ['name' => 'message', 'contents' => $request->message],
        ['name' => 'sandbox', 'contents' => 'false']
    ];

    try {

        $response = Http::asMultipart()
            ->timeout(30)
            ->connectTimeout(10)
            ->retry(2, 2000)
            ->post('https://whatsapp.rajatmarketingss.online/api/create-message', $formData);

        if ($response->successful()) {
            return response()->json([
                'success' => true,
                'message' => 'Message sent successfully.'
            ]);
        }

        // Log API response error for developer
        Log::error('WhatsApp API Failed Response', [
            'status' => $response->status(),
            'body'   => $response->body(),
            'mobile' => $mobileNo
        ]);

        return response()->json([
            'success' => false,
            'message' => 'Message service is temporarily unavailable. Please try again later.'
        ]);

    } catch (ConnectionException $e) {

        // Log detailed error for developer
        Log::error('WhatsApp API Connection Error', [
            'error'  => $e->getMessage(),
            'mobile' => $mobileNo
        ]);

        return response()->json([
            'success' => false,
            'message' => 'Unable to connect to message service. Please try again later.'
        ]);

    } catch (\Exception $e) {

        Log::error('WhatsApp Unexpected Error', [
            'error'  => $e->getMessage(),
            'mobile' => $mobileNo
        ]);

        return response()->json([
            'success' => false,
            'message' => 'Something went wrong while sending message.'
        ]);
    }
}

public function sendOtp(Request $request)
{
    $validator = Validator::make($request->all(), [
        'mobile' => 'required|digits:10'
    ]);
    if ($validator->fails()) {
        return response()->json(['success' => false, 'message' => $validator->errors()->first()]);
    }

    $otp = rand(100000, 999999);
    $sessionId = uniqid('otp_');
    Session::put($sessionId, ['otp' => $otp, 'mobile' => $request->mobile, 'expires_at' => now()->addMinutes(10)]);

    // Send SMS via Twilio
    $sid = env('TWILIO_SID');
    $token = env('TWILIO_TOKEN');
    $client = new Client($sid, $token);
    $client->messages->create(
        '+91' . $request->mobile, 
        [
            'from' => env('TWILIO_PHONE'),
            'body' => "Your OTP for appointment consent is: $otp. Valid for 10 minutes."
        ]
    );

    return response()->json(['success' => true, 'session_id' => $sessionId]);
}


public function verifyOtp(Request $request)
{
    $validator = Validator::make($request->all(), [
        'session_id' => 'required|string',
        'otp' => 'required|digits:6'
    ]);
    if ($validator->fails()) {
        return response()->json(['success' => false, 'message' => $validator->errors()->first()]);
    }

    $sessionData = Session::get($request->session_id);
    if (!$sessionData || now()->gt($sessionData['expires_at']) || $request->otp != $sessionData['otp']) {
        return response()->json(['success' => false, 'message' => 'Invalid or expired OTP']);
    }

    // Optionally extend session or mark verified
    return response()->json(['success' => true]);
  }

                    "📍 Address: $clinicAddress\n";

                $message .= "💰 Consultation Fee: ₹" . number_format($consultationFee, 2) . "\n";
            }
            
            $message .= "\nRegards,\nTeam SkoraCares";

    public function editappointment(Request $request)
    {
        $encryptedId = $request->query('appointment_id');
        if (!$encryptedId) {
            return redirect()->route('doctors.appointment')
                            ->with('error', 'Invalid Request.');
        }

        try {
            $appointmentId = Crypt::decryptString($encryptedId);
        } catch (\Exception $e) {
            return redirect()->route('doctors.appointment')
                            ->with('error', 'Invalid or Tampered Appointment Link.');
        }

        $appointment = Appointment::with('patient')->findOrFail($appointmentId);

        if ($appointment->doctor_id !== auth()->id()) {
            abort(403);
        }

        // Future check
        $apptTime = \Carbon\Carbon::parse($appointment->date . ' ' . $appointment->time);
        if (!$apptTime->isAfter(now())) {
            return redirect()->route('doctors.appointment')
                            ->with('error', 'Past appointments cannot be edited.');
        }

        $clinic = DoctorClinic::where('doctor_id', auth()->id())
                            ->where('is_active', true)
                            ->orderBy('created_at', 'desc')
                            ->first();

        $patient_details = User::where('role', 'patient')
                            ->select('id', 'name', 'phone', 'registration_id', 'gender')
                            ->get();

        return view('doctor.edit-book-appointment', compact('appointment', 'clinic', 'patient_details', 'encryptedId'));
    }


    public function bookedTimes(Request $request)  
    {
        $date = $request->date;
        $excludeId = $request->exclude_id;   

        $booked = Appointment::where('doctor_id', auth()->id())
                    ->where('date', $date)
                    ->when($excludeId, fn($q) => $q->where('id', '!=', $excludeId))
                    ->pluck('time')
                    ->toArray();

        return response()->json(['booked_times' => $booked]);
    }

   public function filter_patients_appointments(Request $request)
    {
        $doctorId = Auth::id();
        
        $query = Appointment::where('doctor_id', $doctorId)->with('patient')->orderBy('date', 'desc')->orderBy('time', 'desc');

        if ($request->filled('status')) {
            if ($request->status === 'pending') {
                $query->where('status', 'pending');
            } else {
                $query->where('status', $request->status); 
            }
        }

        if ($request->filled('name')) {
            $query->whereHas('patient', function ($q) use ($request) {
                $q->where('name', 'like', '%' . $request->name . '%');
            });
        }
        if ($request->filled('phone')) {
            $query->whereHas('patient', function ($q) use ($request) {
                $q->where('phone', 'like', '%' . $request->phone . '%');
            });
        }

        if ($request->filled('start_date') && $request->filled('end_date')) {
            $start = Carbon::parse($request->start_date)->startOfDay();
            $end = Carbon::parse($request->end_date)->endOfDay();
            $query->whereBetween('date', [$start, $end]);
        }

        if (!$request->filled('start_date') && !$request->filled('end_date') && !$request->filled('name') && !$request->filled('phone')) {
            $query->whereDate('date', today());  
        }
        $appointments = $query->paginate(10);
        foreach ($appointments as $key => $appointment) {
            $appointment->index = $appointments->firstItem() + $key;
            
            if ($appointment->patient) {
                $appointment->patient->age = $appointment->patient->dob ? Carbon::parse($appointment->patient->dob)->age : 0;
            }
            
            $appointment->encrypted_id = Crypt::encryptString($appointment->id);
        }

        $todayclinical_visit_count = Appointment::where('doctor_id', $doctorId)
                ->whereDate('date', today())
                ->where('case_type', 'clinical_visit')
                ->where('status', 'pending')
                ->count();

        $todayhome_visit_count = Appointment::where('doctor_id', $doctorId)
                ->whereDate('date', today())
                ->where('case_type', 'home_visit')
                ->where('status', 'pending')
                ->count();

        $todayonline_visit_count = Appointment::where('doctor_id', $doctorId)
                ->whereDate('date', today())
                ->where('case_type', 'online_visit')
                ->where('status', 'pending')
                ->count();

        $todaycall_visit_count = Appointment::where('doctor_id', $doctorId)
                ->whereDate('date', today())
                ->where('case_type', 'on_call_visit')
                ->where('status', 'pending')
                ->count();

        return response()->json([
            'success' => true,
            'appointments' => $appointments->items(),
            'todayclinical_visit_count' => $todayclinical_visit_count,
            'todayhome_visit_count' => $todayhome_visit_count,
            'todayonline_visit_count' => $todayonline_visit_count,
            'todaycall_visit_count' => $todaycall_visit_count,
            'pagination' => [
                'current_page' => $appointments->currentPage(),
                'last_page' => $appointments->lastPage(),
                'total' => $appointments->total(),
                'links' => $appointments->links('pagination::bootstrap-5')->toHtml()
            ]
        ]);
    }


    public function cancelAppointment(Request $request)
    {
        $validator = Validator::make($request->all(), [
            'appointment_id' => 'required|exists:appointments,id',
        ]);

        if ($validator->fails()) {
            return response()->json([
                'success' => false,
                'message' => $validator->errors()->first()
            ], 422);
        }

        $appointment = Appointment::with('patient')->findOrFail($request->appointment_id);
        
        if ($appointment->doctor_id !== auth()->id()) {
            return response()->json([
                'success' => false,
                'message' => 'Unauthorized action'
            ], 403);
        }
        
        $apptTime = Carbon::parse($appointment->date . ' ' . $appointment->time);
        if ($apptTime->isPast()) {
            return response()->json([
                'success' => false,
                'message' => 'Cannot cancel past appointments'
            ], 422);
        }
        
        if ($appointment->status === 'cancelled') {
            return response()->json([
                'success' => false,
                'message' => 'Appointment is already cancelled'
            ], 422);
        }

        if ($appointment->status !== 'pending') {
            return response()->json([
                'success' => false,
                'message' => 'Only Pending appointments can be cancelled'
            ], 422);
        }
        
        $appointment->update(['status' => 'cancelled']);
        
        // WhatsApp notification
        $patient = $appointment->patient;
        $doctor  = auth()->user();
        $patientName = $patient ? $patient->name : $appointment->patient_string;
        $mobileNo    = $patient && $patient->phone ? $patient->phone : $appointment->mobile_number;
        
        if ($mobileNo && !str_starts_with($mobileNo, '+')) {
            $mobileNo = '+91' . $mobileNo;
        }
        
        if ($mobileNo) {
            $clinic = DoctorClinic::where('doctor_id', auth()->id())
                                ->where('is_active', true)
                                ->latest()
                                ->first();

            $clinicName    = $clinic ? $clinic->clinic_name : 'SkoraCares';
            $clinicAddress = $clinic ? $clinic->address : 'Noida (UP), Ghaziabad';

            $message = "Dear {$patientName},\n\n" .
                    "Your appointment with Dr. {$doctor->name} on " .
                    Carbon::parse($appointment->date)->format('l, F d, Y') .
                    " at {$appointment->time} has been **CANCELLED** by the doctor.\n\n" .
                    "If this was unintentional, please contact us immediately.\n\n" .
                    "Clinic: {$clinicName}\nAddress: {$clinicAddress}\n\n" .
                    "Regards,\nTeam SkoraCares";

            try {

    public function completeAppointment(Request $request)
    {
        $validator = Validator::make($request->all(), [
            'appointment_id' => 'required|exists:appointments,id',
        ]);

        if ($validator->fails()) {
            return response()->json([
                'success' => false,
                'message' => $validator->errors()->first()
            ], 422);
        }

        $appointment = Appointment::findOrFail($request->appointment_id);
        
        if ($appointment->doctor_id !== auth()->id()) {
            return response()->json([
                'success' => false,
                'message' => 'Unauthorized action'
            ], 403);
        }
        
        if ($appointment->status === 'completed') {
            return response()->json([
                'success' => false,
                'message' => 'Appointment is already completed'
            ], 422);
        }
        
        if ($appointment->status === 'cancelled') {
            return response()->json([
                'success' => false,
                'message' => 'Cannot complete cancelled appointments'
            ], 422);
        }
        
        // if ($appointment->status !== 'pending') {
        //     return response()->json([
        //         'success' => false,
        //         'message' => 'Only confirmed appointments can be marked completed'
        //     ], 422);
        // }

         if ($appointment->status !== 'pending') {
        return response()->json([
            'success' => false,
            'message' => 'Only pending appointments can be marked completed'
        ], 422);
    }



        
        $appointment->update(['status' => 'completed']);

        return response()->json([
            'success' => true,
            'message' => 'Appointment marked as completed!'
        ]);
    }

   

    public function show_appointment($id)
    {
        try {
            $appointment = Appointment::with('patient')
                ->where('doctor_id', Auth::id())
                ->findOrFail($id);

            if ($appointment->patient) {
                $appointment->patient->age = $appointment->patient->dob ? Carbon::parse($appointment->patient->dob)->age : 0;
            }

            return response()->json([
                'success' => true,
                'appointment' => $appointment
            ]);

            return response()->json([
                'success' => false,
                'message' => 'Appointment not found.'
            ], 404);
        }
    }

        function parseTime12To24(timeStr) {
            const [time, period] = timeStr.split(' ');
            const [hh, mm] = time.split(':').map(n => parseInt(n, 10));
            let h = hh % 12;
            if (period.toUpperCase() === 'PM' && hh !== 12) h += 12;
            if (period.toUpperCase() === 'AM' && hh === 12) h = 0;
            return { h, m: mm };
        }

        function generateSlots(start, end, interval) {
            const slots = [];
            const nowMs = Date.now();
            const baseDateStr = selectedDate || ymdFormatter.format(new Date());
            const isToday = baseDateStr === ymdFormatter.format(new Date());

            let currentMs = new Date(`${baseDateStr}T${start}:00${IST_OFFSET}`).getTime();
            const endMs = new Date(`${baseDateStr}T${end}:00${IST_OFFSET}`).getTime();

            while (currentMs <= endMs) {
                const timeParts = timeFormatter.formatToParts(new Date(currentMs));
                const hour = timeParts.find(p => p.type === 'hour').value.padStart(2, '0');
                const minute = timeParts.find(p => p.type === 'minute').value;
                const period = timeParts.find(p => p.type === 'dayPeriod') ? timeParts.find(p => p.type === 'dayPeriod').value.toUpperCase() : '';
                const timeStr = `${hour}:${minute} ${period}`;
                const isPast = isToday && currentMs < nowMs;
                slots.push({ time: timeStr, isPast });
                currentMs += interval * 60 * 1000;
            }
            return slots;
        }

        function sendMessage() {
            const mobile = $('#mobileNumber').val();
            if (!mobile || mobile.length !== 10) {
                alert('Please enter a valid 10-digit mobile number.');
                return;
            }
            mobileNumber = mobile;
            const message = $('#messageText').val();

            const $btn = $('#sendMessageBtn');
            const originalText = $btn.html();
            $btn.html('<i class="fas fa-spinner fa-spin"></i> Sending...').prop('disabled', true);

            $.ajax({
                url: '{{ route("doctor.send-whatsapp") }}',
                method: 'POST',
                data: {
                    _token: '{{ csrf_token() }}',
                    mobile: mobile,
                    message: message
                },
                success: function(data) {
                    $btn.html(originalText).prop('disabled', false);
                    if (data.success) {
                        $('#messageStatus').html('<div class="alert alert-success">Message sent successfully via WhatsApp!</div>');
                        $('#messageSent').val('sent');
                    } else {
                        alert('Error sending message: ' + data.message);
                    }
                },
                error: function() {
                    $btn.html(originalText).prop('disabled', false);
                    alert('Error sending message.');
                }
            });
        }

        function sendConsent() {
            const mobile = $('#mobileNumberConsent').val() || mobileNumber;
            if (!mobile || mobile.length !== 10) {
                alert('Please enter a valid 10-digit mobile number.');

            mobileNumber = mobile;

        function previewFile(e) {
            const file = e.target.files[0];
            const preview = $("#filePreview").html('');
            if (file) {
                if (file.type.startsWith("image/")) {
                    const reader = new FileReader();
                    reader.onload = function(ev) {
                        preview.html(`<img src="${ev.target.result}" class="img-fluid rounded shadow file-preview">`);
                    };
                    reader.readAsDataURL(file);
                } else if (file.type === "application/pdf") {
                    preview.html(`<embed src="${URL.createObjectURL(file)}" width="250" height="200" type="application/pdf" class="file-preview">`);

        $(document).on('click', '#submitAppointment', function() {
            const patient = $('#patientSearch').val();
            const patientId = $('#patientId').val();

        function submitAppointmentForm() {
            const patient = $('#patientSearch').val() || '';
            const patientId = $('#patientId').val() || '';
            const remarks = $('#remarks').val() || '';
            const bloodGroup = $('#bloodGroup').val() || '';
            const bp = $('#bp').val() || '';
            const weight = $('#weight').val() || '';
            const height = $('#height').val() || '';
            const caseType = $('#caseType').val() || '';
            const mobile = $('#mobileNumber').val() || mobileNumber || '';

            let isConsentValid = true;
            let consentValue = '';

            if (currentConsentType === 'otp') {
                const sent = $('#messageSent').val();
                if (sent !== 'sent') {
                    alert('Please send the message first.');
                    isConsentValid = false;
                } else {
                    consentValue = 'sent';
                }
            } else if (currentConsentType === 'consent') {
                const sent = $('#consentSent').val();
                if (sent !== 'sent') {
                    alert('Please send the consent link first.');
                    isConsentValid = false;
                } else {
                    consentValue = 'sent';
                }
            } else if (currentConsentType === 'upload') {
                const file = $('#consentFile')[0].files[0];
                if (!file) {
                    alert('Please upload a file.');
                    isConsentValid = false;
                }
            } else if (currentConsentType === 'skipped') {
                consentValue = null;
            }

            if (!isConsentValid) return;

            const $submitButton = $('#submitAppointment');
            const originalButtonText = $submitButton.html();
            $submitButton.html('<i class="fas fa-spinner fa-spin"></i> Booking Appointment...').prop('disabled', true);

            const formData = new FormData();
            formData.append('_token', '{{ csrf_token() }}');
            formData.append('doctor_id', '{{ Auth::id() }}');
            formData.append('patient_id', patientId);
            formData.append('patient_string', patient);
            formData.append('date', selectedDate || '');
            formData.append('time', selectedSlot || '');
            formData.append('case_type', caseType);
            formData.append('blood_group', bloodGroup);
            formData.append('bp', bp);
            formData.append('weight', weight);
            formData.append('height', height);
            formData.append('remarks', remarks);
            formData.append('consent_type', currentConsentType || 'skipped');
            if (consentValue) formData.append('consent_value', consentValue);
            if (currentConsentType === 'upload') {
                const file = $('#consentFile')[0].files[0];
                if (file) formData.append('consent_file', file);
            }
            if (mobile) formData.append('mobile_number', mobile);

            $.ajax({
                url: '{{ route("doctor.appointments.store") }}',
                method: 'POST',
                data: formData,
                processData: false,
                contentType: false,
                success: function(data) {
                    $submitButton.html(originalButtonText).prop('disabled', false);
                    if (data.success) {
                        showAlert('Appointment booked successfully!', 'success');
                        $('#appointmentModal').modal('hide');
                        selectedSlot = null;
                        loadAllSlots();
                        currentConsentType = null;
                    } else {
                        showAlert('Error booking appointment: ' + (data.message || 'Unknown error'), 'error');
                    }
                },
                error: function(xhr) {
                    $submitButton.html(originalButtonText).prop('disabled', false);
                    const err = xhr.responseJSON;
                    showAlert('Error occurred: ' + (err.message || 'Server error'), 'error');
                }
            });
        }

        toggleConsent('otp');
        $('input[name="consent"]').on('change', function(){ toggleConsent($(this).val()); });
        loadAllSlots();

        $(".toggle-address-details").on("click", function(){
            $(".address-details").slideToggle();
        });
    });
    </script>
@endsection

    });

    // Search with debounce
    let searchTimer;
    $('#searchName, #searchPhone').on('keyup', function() {
        clearTimeout(searchTimer);
        searchTimer = setTimeout(() => {
            currentPage = 1;
            filterData();
        }, 500);
    });

    // Filter data function
    function filterData() {
        let data = {
            page: currentPage,
            status: currentStatus,
            name: $('#searchName').val().trim(),
            phone: $('#searchPhone').val().trim()
        };

        if (selectedStartDate && selectedEndDate) {
            data.start_date = selectedStartDate;
            data.end_date = selectedEndDate;
        }

        $('#appointmentTableBody').html(`
            <tr>
                <td colspan="8" class="text-center py-4">
                    <i class="fas fa-spinner fa-spin fa-2x text-primary"></i>
                    <p class="mt-2">Loading appointments...</p>
                </td>
            </tr>
        `);

        $.ajax({
            url: '{{ route("doctor.filter_patients_appointments") }}',
            type: 'GET',
            data: data,
            success: function(response) {
                if (response.success) {
                    renderTable(response);
                    updateCounts(response);
                } else {
                    showError('Failed to load appointments');
                }
            },
            error: function() {
                showError('Error loading data');
            }
        });
    }

    // Render table
    function renderTable(response) {
        let html = '';
        
        if (response.appointments && response.appointments.length > 0) {
            response.appointments.forEach(function(appointment) {
                const patient = appointment.patient || {};
                const profileImage = patient.profile_image 
                    ? '{{ asset('storage/') }}' + patient.profile_image 
                    : '{{ asset('assets-doctor/img/profiles/avatar-01.jpg') }}';
                
                const apptDatetime = moment(appointment.date + ' ' + appointment.time, 'YYYY-MM-DD hh:mm A');
                const isFuture = apptDatetime.isAfter(moment());
                const isConfirmed = (appointment.status === 'confirmed' || currentStatus === 'pending');

                let editHtml = '';
                let completeHtml = '';
                let cancelHtml = '';

                if (isConfirmed) {
                    if (isFuture) {
                        editHtml = `<li><a href="{{ route('edit-book-appointment') }}?appointment_id=${appointment.encrypted_id}" class="dropdown-item">Edit Appointment</a></li>`;
                    }
                    
                    completeHtml = `<li><a href="#" class="dropdown-item complete-appointment" data-id="${appointment.id}" data-name="${patient.name || ''}">Mark Completed</a></li>`;
                    
                    if (isFuture) {
                        cancelHtml = `<li><a href="#" class="dropdown-item cancel-appointment" data-id="${appointment.id}" data-name="${patient.name || ''}" data-phone="${patient.phone || ''}" data-gender="${patient.gender || ''}" data-age="${patient.age || ''}" data-regid="${patient.registration_id || ''}">Cancel Appointment</a></li>`;
                    }
                }

                html += `
                <tr>
                    <td>${appointment.index}</td>
                    <td>
                        <div class="d-flex align-items-center">
                            <img src="${profileImage}" class="rounded-circle me-2" style="width: 40px; height: 40px; object-fit: cover;">
                            <div>
                                <strong>${patient.name || 'N/A'}</strong><br>
                                <small>${patient.gender || ''} ${patient.age || ''}y</small>
                            </div>
                        </div>
                    </td>
                    <td>${patient.phone || appointment.mobile_number || 'N/A'}</td>
                    <td>${patient.registration_id || ''}</td>
                    <td>
                        <span class="badge ${getVisitTypeClass(appointment.case_type)}">
                            ${appointment.case_type ? appointment.case_type.replace('_', ' ') : 'N/A'}
                        </span>
                    </td>
                    <td>
                        ${moment(appointment.date).format('DD MMM YYYY')}<br>
                        <small>${appointment.time}</small>
                    </td>
                    <td>
                        ${isConfirmed ? 
                            `<a href="{{ url('/doctor-consultation') }}/${appointment.patient_id}" class="btn btn-sm btn-outline-primary">Consult</a>` :
                            `<span class="badge ${getStatusBadgeClass(appointment.status)}">${appointment.status}</span>`
                        }
                    </td>
                    <td>
                        <div class="dropdown">
                            <a href="#" class="text-muted" data-bs-toggle="dropdown">
                                <i class="ti ti-dots-vertical"></i>
                            </a>
                            <ul class="dropdown-menu">
                                <li><a class="dropdown-item view-appointment" href="#" data-id="${appointment.id}">View Details</a></li>
                                ${editHtml}
                                ${completeHtml}
                                ${cancelHtml}
                            </ul>
                        </div>
                    </td>
                </tr>`;
            });
        } else {
            html = `
            <tr>
                <td colspan="8" class="text-center py-5">
                    <i class="ti ti-calendar-x fs-48 text-muted mb-3"></i>
                    <h5 class="text-muted">No appointments found</h5>
                </td>
            </tr>`;
        }

        $('#appointmentTableBody').html(html);
        $('#pagination').html(response.pagination?.links || '');
    }

    // Helper functions
    function getVisitTypeClass(type) {
        const classes = {
            'clinical_visit': 'bg-primary',
            'home_visit': 'bg-success',
            'online_visit': 'bg-info',
            'on_call_visit': 'bg-secondary'
        };
        return classes[type] || 'bg-secondary';
    }

    function getStatusBadgeClass(status) {
        const classes = {
            'confirmed': 'bg-primary',
            'completed': 'bg-success',
            'cancelled': 'bg-danger'
        };
        return classes[status] || 'bg-secondary';
    }

    function updateCounts(data) {
        $('#today-clinical-visit').text(data.todayclinical_visit_count || 0);
        $('#today-home-visit').text(data.todayhome_visit_count || 0);
        $('#today-online-visit').text(data.todayonline_visit_count || 0);
        $('#today-call-visit').text(data.todaycall_visit_count || 0);
    }

    function showError(message) {
        $('#appointmentTableBody').html(`
            <tr>
                <td colspan="8" class="text-center py-4 text-danger">
                    <i class="ti ti-alert-circle fs-48 mb-3"></i>
                    <p>${message}</p>
                </td>
            </tr>
        `);
    }

    function showAlert(message, type) {
        if (typeof window.showAlert === 'function') {
            window.showAlert(message, type);
        } else {
            alert(message);
        }
    }

    // Pagination
    $(document).on('click', '.pagination a', function(e) {
        e.preventDefault();
        let url = new URL($(this).attr('href'));
        currentPage = url.searchParams.get('page') || 1;
        filterData();
    });

    // View appointment
    $(document).on('click', '.view-appointment', function(e) {
        e.preventDefault();
        let id = $(this).data('id');
        
        $.ajax({
            url: '/doctor-appointment/' + id,
            type: 'GET',
            success: function(response) {
                if (response.success) {
                    let appt = response.appointment;

                    $('#modal-profile-image').attr('src', 
                        patient.profile_image

                            : '{{ asset('assets-doctor/img/profiles/avatar-01.jpg') }}'
                    );
                    $('#modal-name').text('Patient: ' + (patient.name || 'N/A'));
                    $('#modal-patient-id').text('ID: ' + (patient.registration_id || 'N/A'));
                    $('#modal-age').text(patient.age || 'N/A');
                    $('#modal-gender').text(patient.gender || 'N/A');
                    $('#modal-phone').text(patient.phone || 'N/A');
                    $('#modal-blood-group').text(appt.blood_group || 'N/A');
                    $('#modal-date').text(moment(appt.date).format('DD MMM YYYY'));
                    $('#modal-time').text(appt.time || 'N/A');
                    $('#modal-visit-type').text(appt.case_type ? appt.case_type.replace('_', ' ') : 'N/A');
                    
                    let statusClass = getStatusBadgeClass(appt.status);
                    $('#modal-status').text(appt.status).removeClass().addClass(`badge ${statusClass}`);
                    
                    $('#modal-bp').text(appt.bp || 'N/A');
                    $('#modal-weight').text(appt.weight || 'N/A');
                    $('#modal-height').text(appt.height || 'N/A');
                    $('#modal-remarks').text(appt.remarks || 'No remarks');
                    $('#modal-note').text(appt.note || 'No notes');
                    
                    $('#viewModal').modal('show');
                }
            }
        });
    });

    // Cancel appointment
    $(document).on('click', '.cancel-appointment', function(e) {
        e.preventDefault();
        currentCancelId = $(this).data('id');
        $('#cancelPatientName').text(
            $(this).data('name') + ' (' + $(this).data('gender') + ', ' + $(this).data('age') + 'y)'
        );
        $('#cancelPatientPhone').text($(this).data('phone') || 'N/A');
        $('#cancelPatientRegId').text('ID: ' + ($(this).data('regid') || 'N/A'));
        $('#cancelAppointmentModal').modal('show');
    });

    $('#confirmCancelBtn').on('click', function() {
        if (!currentCancelId) return;
        
        const $btn = $(this).prop('disabled', true).html('<i class="fas fa-spinner fa-spin"></i> Cancelling...');
        
        $.ajax({
            url: '{{ route("doctor.appointment.cancel") }}',
            method: 'POST',
            data: {
                _token: '{{ csrf_token() }}',
                appointment_id: currentCancelId
            },
            success: function(res) {
                if (res.success) {
                    $('#cancelAppointmentModal').modal('hide');
                    showAlert(res.message, 'success');
                    filterData();
                } else {
                    showAlert(res.message, 'error');
                }
            },
            error: function(xhr) {
                showAlert(xhr.responseJSON?.message || 'Error', 'error');
            },
            complete: function() {
                $btn.prop('disabled', false).html('Yes, Cancel');
                currentCancelId = null;
            }
        });
    });

    // Complete appointment
    $(document).on('click', '.complete-appointment', function(e) {
        e.preventDefault();
        currentCompleteId = $(this).data('id');
        $('#completePatientName').text($(this).data('name'));
        $('#completeAppointmentModal').modal('show');
    });

    $('#confirmCompleteBtn').on('click', function() {
        if (!currentCompleteId) return;
        
        const $btn = $(this).prop('disabled', true).html('<i class="fas fa-spinner fa-spin"></i> Processing...');
        
        $.ajax({
            url: '{{ route("doctor.appointment.complete") }}',
            method: 'POST',
            data: {
                _token: '{{ csrf_token() }}',
                appointment_id: currentCompleteId
            },
            success: function(res) {
                if (res.success) {
                    $('#completeAppointmentModal').modal('hide');
                    showAlert(res.message, 'success');
                    filterData();
                } else {
                    showAlert(res.message, 'error');
                }
            },
            error: function(xhr) {
                showAlert(xhr.responseJSON?.message || 'Error', 'error');
            },
            complete: function() {
                $btn.prop('disabled', false).html('Yes, Complete');
                currentCompleteId = null;
            }
        });
    });

    // Clear modals
    $('#cancelAppointmentModal').on('hidden.bs.modal', function() {
        currentCancelId = null;
    });

    $('#completeAppointmentModal').on('hidden.bs.modal', function() {
        currentCompleteId = null;
    });

    // Initial load
    filterData();
});
</script>
