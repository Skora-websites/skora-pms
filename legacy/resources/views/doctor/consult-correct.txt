            function saveConsultation(callback) {
                const consultationData = {
                    patient_id: {{ $patient->id ?? 'null' }},
                    appointment_id: "{{ $appointments->id ?? '' }}",
                    symptoms_note: $('#symptomsNote').val(),
                    examination_note: $('#examinationNote').val(),

                    examination: getTableData('#examinationTable'),
                    diagnosis: getTableData('#diagnosisTable'),
                    lab_tests: getTableData('#labTestsTable'),
                    medications: getMedicationsData(),
                };
                $.ajax({
                    url: "{{ route('consultations.store') }}",
                    method: 'POST',
                    data: {
                        _token: '{{ csrf_token() }}',
                        ...consultationData
                    },

                    },

                    }
                });

            // Voice Recognition Code (same, no change)
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            let recognition;
            let currentTargetId = '';
            let fullTranscript = '';
            const voiceModal = $('#voiceModal');
            const transcriptPreview = $('#transcriptPreview');
            const stopBtn = $('#stopVoiceBtn');
            const closeBtn = $('#closeModalBtn');
            if (SpeechRecognition) {
                recognition = new SpeechRecognition();
                recognition.lang = 'en-US';
                recognition.continuous = true;
                recognition.interimResults = true;
                recognition.onresult = (event) => {
                    let interim = '';
                    let final = '';
                    for (let i = event.resultIndex; i < event.results.length; ++i) {
                        if (event.results[i].isFinal) {
                            final += event.results[i][0].transcript + ' ';
                        } else {
                            interim = event.results[i][0].transcript;
                        }
                    }
                    fullTranscript += final;
                    const currentText = fullTranscript + (interim ? ' ' + interim : '');
                    transcriptPreview.text(currentText || 'Listening... Speak now.');
                    $('#' + currentTargetId).val(currentText); // Update textarea with interim too
                };
                // In onerror, handle 'no-speech' specifically
                recognition.onerror = (event) => {
                    console.error('Speech recognition error:', event.error);
                    if (event.error === 'no-speech') {
                        transcriptPreview.text('No speech detected. Speak louder or check mic. Try again.');
                    } else if (event.error === 'not-allowed') {
                        transcriptPreview.text('Microphone access denied. Please allow microphone access in browser settings.');
                    } else if (event.error === 'service-not-allowed') {
                        transcriptPreview.text('Speech service not allowed. Check browser permissions.');
                    } else {
                        transcriptPreview.text('Error: ' + event.error + '. Try again.');
                    }
                    setTimeout(() => {
                        if (!stopBtn.data('stopped')) recognition.start();
                    }, 1000);
                    $(`.voice-mic[data-target="${currentTargetId}"]`).siblings('.voice-indicator').text('');
                    setTimeout(() => closeVoiceModal(), 5000);
                };
                recognition.onend = () => {
                    if (!stopBtn.data('stopped')) {
                        recognition.start();
                    } else {
                        $(`.voice-mic[data-target="${currentTargetId}"]`).siblings('.voice-indicator').text('');
                    }
                };
                $('.voice-mic').on('click', function() {
                    currentTargetId = $(this).data('target');
                    fullTranscript = '';
                    transcriptPreview.text('Listening... Speak now.');
                    voiceModal.css('display', 'flex');
                    stopBtn.data('stopped', false);
                    const input = $('#' + currentTargetId);
                    input.val('');
                    recognition.start();
                });
                stopBtn.on('click', function() {
                    $(this).data('stopped', true);
                    recognition.stop();
                    closeVoiceModal();
                });
                closeBtn.on('click', function() {
                    recognition.stop();
                    closeVoiceModal();
                });
                function closeVoiceModal() {
                    voiceModal.css('display', 'none');
                    $(`.voice-mic[data-target="${currentTargetId}"]`).siblings('.voice-indicator').text('');
                    currentTargetId = '';
                    fullTranscript = '';
                }
            } else {
                $('.voice-mic').hide();
                showAlert('Speech Recognition not supported in this browser. Please use Chrome or Edge.',
                    'error');
            }
