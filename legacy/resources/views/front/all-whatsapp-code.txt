 public function sendOtp(Request $request)
{

// for Extra security code ok
//   $key = 'otp_request_' . $request->ip();
//     $requests = Cache::get($key, 0);
    
//     if ($requests >= 3) {

//         ], 429);

//     Cache::put($key, $requests + 1, now()->addMinutes(10));




    $request->validate(['mobile' => 'required|digits:10']);
    $mobile = $request->mobile;
    $fullMobile = '+91' . $mobile; 
    $otp = rand(100000, 999999);
    Cache::put("otp_{$mobile}", $otp, now()->addMinutes(5));
    Log::info("OTP for {$fullMobile}: {$otp}");

    $message = "Dear Patient,\nYour OTP for registration is: {$otp}\nThis OTP is valid for 5 minutes.";
    try {
        $formData = [
            [
                'name' => 'appkey',
                'contents' => env('WHATSAPP_APP_KEY', $this->appKey) 
            ],
            [
                'name' => 'authkey',
                'contents' => env('WHATSAPP_AUTH_KEY', $this->authKey) 
            ],
            [
                'name' => 'to',
                'contents' => $fullMobile 
            ],
            [
                'name' => 'message',
                'contents' => $message
            ],
            [
                'name' => 'sandbox',
                'contents' => 'false'
            ]
        ];
        $response = Http::asMultipart()
            ->timeout(30) 
            ->post('https://whatsapp.rajatmarketingss.online/api/create-message', $formData);
        \Log::info('WhatsApp API Request: ', $formData);
        \Log::info('WhatsApp API response: status=' . $response->status() . ', body=' . $response->body());

        if ($response->successful()) {
            $responseData = $response->json();
            if (isset($responseData['message_status']) && $responseData['message_status'] === 'Success') {
                return response()->json([
                    'success' => true, 
                    'message' => 'OTP sent successfully via WhatsApp'
                ]);
            } else {
                // \Log::error('WhatsApp API returned error: ' . $response->body());
                return response()->json([
                    'success' => false, 
                    'message' => 'Failed to send OTP via WhatsApp'
                ], 500);
            }
        } else {
            // \Log::error('WhatsApp API failed: ' . $response->body());
            return response()->json([
                'success' => false, 
                'message' => 'WhatsApp service temporarily unavailable'
            ], 500);
        }
        
    } catch (\Exception $e) {
        // \Log::error('Error sending WhatsApp message: ' . $e->getMessage());
        return response()->json([
            'success' => false, 
            'message' => 'Error sending OTP: ' . $e->getMessage()
        ], 500);
    }
}

    
    public function verifyOtp(Request $request)
    {
        $key = 'otp_attempts_' . $request->ip();
        $attempts = Cache::get($key, 0);
        if ($attempts >= 5) {
            return response()->json([
                'success' => false, 
                'message' => 'Too many attempts. Try again after 15 minutes.'
            ], 429);
        }
        
        $mobile = $request->mobile;
        $cachedOtp = Cache::get("otp_{$mobile}");

        if ($cachedOtp && $cachedOtp == $request->otp) {
            Cache::forget("otp_{$mobile}");
            Cache::forget($key); // Reset attempts on success
            return response()->json(['success' => true, 'message' => 'OTP verified successfully']);
        }
        Cache::put($key, $attempts + 1, now()->addMinutes(15));
        return response()->json(['success' => false, 'message' => 'Invalid OTP'], 400);
    }

public function register(Request $request)
{
    // dd($request->all());
    $request->validate([
        'name' => 'required|string|max:255',
        'mobile' => 'required|digits:10',
        'email' => 'required|email|unique:users,email',
        'password' => 'required|string|min:8|confirmed', 
        'role' => 'required|in:doctor,patient',
        'gender' => 'nullable|in:Male,Female,Other',

        // Optional fields
        'clinic_name' => 'nullable|string|max:255',
        'degrees' => 'nullable|array|max:100',
        'specializations' => 'nullable|array|max:100',
        'country' => 'nullable|string|max:255',
        'state' => 'nullable|string|max:255',
        'city' => 'nullable|string|max:255',
        'address' => 'nullable|string|max:500',
        'current_address' => 'nullable|string|max:500',
    ]);

    try {
        $mobile = $request->mobile;

        $data = $request->only([
            'name','email','password',
            'clinic_name','country','state','city','address',
            'current_address','role','gender'
        ]);

        // convert arrays into comma separated strings
        $data['degrees'] = $request->degrees ? implode(',', $request->degrees) : null;
        $data['specializations'] = $request->specializations ? implode(',', $request->specializations) : null;

        $data['phone'] = $mobile;
        $data['password'] = Hash::make($request->password);

        // Create user
        $user = User::create($data);

        // Dispatch WhatsApp notification (optional)
        // SendRegistrationWhatsApp::dispatch($mobile, $request->name, $request->email);

        // Login user
        auth()->login($user);

        $redirectUrl = $user->role === 'patient' 
            ? route('patient.dashboard')   // named route (recommended)
            : route('doctor.dashboard');   // ya phir direct '/doctor-dashboard'

        return response()->json([
            'success' => true,
            'message' => 'Registration successfully!',
            'redirect' => $redirectUrl
        ]);
    } catch (\Exception $e) {
        Log::error('Registration failed: ' . $e->getMessage(), ['request' => $request->all()]);
        return response()->json([
            'success' => false,
            'message' => 'Registration failed. Please try again.'
        ], 500);
    }
}

        // Custom JS to ensure smooth slow transitions on carousel without disappearing/gaps, and immediate next show at interval
        document.addEventListener('DOMContentLoaded', function() {
            var carouselElement = document.querySelector('#loginCarousel');
            
            // Force preload all images immediately
            var images = carouselElement.querySelectorAll('img');
            images.forEach(function(img) {
                if (img.complete) {
                    img.style.opacity = '1';
                } else {
                    img.addEventListener('load', function() {
                        this.style.opacity = '1';
                    });
                }
            });

            var bootstrapCarousel = new bootstrap.Carousel(carouselElement, {
                interval: 12000, // Reinforce the slow interval
                ride: 'carousel',
                wrap: true
            });

            // Listen to slide events to ensure seamless overlap - no gaps, immediate next visibility at interval start
            carouselElement.addEventListener('slide.bs.carousel', function (e) {
                // Immediately set the next item to full opacity to start showing without delay
                var nextItem = carouselElement.querySelector('.carousel-item-next');
                if (nextItem) {
                    nextItem.style.opacity = '1'; // Immediate show for next
                    nextItem.style.zIndex = '2'; // Higher z-index for overlap during fade
                }
                // Slow fade out for current active
                var activeItem = carouselElement.querySelector('.carousel-item.active');
                if (activeItem) {
                    activeItem.style.transition = 'opacity 3s ease-in-out';
                    activeItem.style.opacity = '0'; // Start slow fade out
                }
            });

            carouselElement.addEventListener('slid.bs.carousel', function (e) {
                // Reset z-index and ensure full setup after transition
                var activeItem = carouselElement.querySelector('.carousel-item.active');
                if (activeItem) {
                    activeItem.style.opacity = '1';
                    activeItem.style.zIndex = '1';
                }
                var nextItem = carouselElement.querySelector('.carousel-item-next');
                if (nextItem) {
                    nextItem.style.zIndex = '0';
                }
                // Force reflow to apply styles
                carouselElement.offsetHeight;
            });
        });

        // Theme Toggle Function
        function toggleTheme() {
            const body = document.body;
            const currentTheme = body.getAttribute('data-theme');
            const newTheme = currentTheme === 'light' ? 'dark' : 'light';
            body.setAttribute('data-theme', newTheme);
            const icon = document.querySelector('.theme-toggle i');
            icon.classList.toggle('bi-moon-stars-fill');
            icon.classList.toggle('bi-sun-fill');
            localStorage.setItem('theme', newTheme);
        }

        // Load saved theme
        document.addEventListener('DOMContentLoaded', () => {
            const savedTheme = localStorage.getItem('theme') || 'light';
            document.body.setAttribute('data-theme', savedTheme);
            const icon = document.querySelector('.theme-toggle i');
            if (savedTheme === 'dark') {
                icon.classList.remove('bi-moon-stars-fill');
                icon.classList.add('bi-sun-fill');
            }
        });

        document.addEventListener("DOMContentLoaded", function () {

    <script>
        $(document).ready(function() {
            $('.js-example-basic-multiple').select2({
                placeholder: "Search and select items...",
                allowClear: true,
                closeOnSelect: false,
                templateSelection: function(data) {
                    return data.text || data.id;
                }
            });
            $('.js-example-basic-multiple').on('change', function() {
                let selectedValues = $(this).val();
                console.log('Selected:', selectedValues);
            });

            // Initialize Google Places Autocomplete
            let autocomplete;
            function initAutocomplete() {
                autocomplete = new google.maps.places.Autocomplete(
                    document.getElementById('current_address'),
                    { types: ['geocode'] }
                );
                autocomplete.addListener('place_changed', function() {
                    const place = autocomplete.getPlace();
                    if (place.formatted_address) {
                        $('#current_address').val(place.formatted_address);
                    }
                });
            }

            // Initialize autocomplete when Google Maps API is loaded
            if (typeof google !== 'undefined' && google.maps) {
                initAutocomplete();
            } else {
                window.addEventListener('load', initAutocomplete);
            }
        });

        const formSteps = document.querySelectorAll('.form-step');
        const nextBtns = document.querySelectorAll('.next-btn');
        const prevBtns = document.querySelectorAll('.prev-btn');
        const skipBtns = document.querySelectorAll('.skip-btn');
        const stepProgressContainer = document.getElementById('stepProgressContainer');
        const initialProgressHtml = stepProgressContainer.innerHTML;
        let stepCircles = document.querySelectorAll('.step-circle');
        let stepProgress = document.querySelector('.step-progress');
        let currentStep = 0;
        let isOtpVerified = false;
        let selectedRole = '';
        let isPatient = false;

        function showStep(step) {
            let formIndex = step;
            if (step >= 2) {
                if (isPatient) {
                    formIndex = 5;
                } else {
                    formIndex = step;
                }
            }
            formSteps.forEach((s, i) => s.classList.toggle('d-none', i !== formIndex));
            stepProgress = document.querySelector('.step-progress');
            stepProgress.className = `step-progress step-${step} ${isPatient ? 'three-steps' : 'five-steps'}`;
            stepCircles = document.querySelectorAll('.step-circle');
            stepCircles.forEach((circle, i) => circle.classList.toggle('active', i <= step));
        }

        function showLoader(show) {
            $('#loadingOverlay').css('display', show ? 'flex' : 'none');
        }

        function showNotification(msg, type = 'success') {
            let alertClass = `alert-${type}`;
            let iconClass = '';
            let textClass = '';
            switch (type) {
                case 'success':
                    iconClass = 'fas fa-check-circle text-success';
                    textClass = 'text-success';
                    break;
                case 'error':
                    iconClass = 'fas fa-exclamation-circle text-danger';
                    textClass = 'text-danger';
                    break;
                case 'info':
                    iconClass = 'fas fa-info-circle text-info';
                    textClass = 'text-info';
                    break;
                case 'warning':
                    iconClass = 'fas fa-exclamation-triangle text-warning';
                    textClass = 'text-warning';
                    break;
                default:
                    iconClass = 'fas fa-check-circle text-success';
                    textClass = 'text-success';
            }
            var alertBox = document.createElement("div");
            alertBox.className = `custom-alert-box ${alertClass} notification-sidebar position-fixed top-2 mt-3 shadow-lg rounded`;
            alertBox.innerHTML = `
                <div class="${textClass} p-custom">
                    <i class="${iconClass} icon"></i>
                    ${msg}
                    <button type="button" class="close-btn" onclick="this.parentElement.parentElement.remove()">&times;</button>
                </div>
            `;
            document.body.appendChild(alertBox);
            setTimeout(() => {
                alertBox.classList.add('show-notification');
            }, 10);
            setTimeout(() => {
                alertBox.style.transition = "right 0.5s ease-in-out, opacity 0.5s ease";
                alertBox.style.opacity = "0";
                setTimeout(() => alertBox.remove(), 500);
            }, 8000);
        }

        $('#sendOtpBtn').on('click', function() {
            const mobile = $('#mobile').val();
            if (!mobile) {
                showNotification('Please enter mobile number', 'error');
                return;
            }
            $(this).prop('disabled', true);
            showLoader(true);
            $.ajax({
                url: '{{ route("otp.send") }}',
                type: 'POST',
                data: { mobile: mobile, _token: '{{ csrf_token() }}' },
                success: function(response) {
                    showLoader(false);
                    $('#sendOtpBtn').prop('disabled', false);
                    if (response.success) {
                        showNotification(response.message, 'success');
                        $('#otpSection').removeClass('d-none');
                        $('#sendOtpBtn').addClass('d-none');
                        $('#verifyOtpBtn').removeClass('d-none');
                    } else {
                        showNotification(response.message || 'Failed to send OTP', 'error');
                    }
                },
                error: function() {
                    showLoader(false);
                    $('#sendOtpBtn').prop('disabled', false);
                    showNotification('Error sending OTP', 'error');
                }
            });
        });

        $('#verifyOtpBtn').on('click', function() {
            const mobile = $('#mobile').val();
            const otp = $('#otp').val();
            if (!otp) {
                showNotification('Please enter OTP', 'error');
                return;
            }
            $(this).addClass('loading').prop('disabled', true);
            showLoader(true);
            $.ajax({
                url: '{{ route("otp.verify") }}',
                type: 'POST',
                data: { mobile: mobile, otp: otp, _token: '{{ csrf_token() }}' },
                success: function(response) {
                    showLoader(false);
                    $('#verifyOtpBtn').removeClass('loading').prop('disabled', false);
                    if (response.success) {
                        showNotification(response.message, 'success');
                        isOtpVerified = true;
                        currentStep = 1;
                        showStep(currentStep);
                    } else {
                        showNotification(response.message || 'Invalid OTP', 'error');
                    }
                },
                error: function() {
                    showLoader(false);
                    $('#verifyOtpBtn').removeClass('loading').prop('disabled', false);
                    showNotification('Error verifying OTP', 'error');
                }
            });

        $('#useCurrentLocation').on('click', function() {
            if (!navigator.geolocation) {
                showNotification('Geolocation is not supported by your browser', 'error');
                return;
            }
            $(this).addClass('loading').prop('disabled', true);
            showLoader(true);
            navigator.geolocation.getCurrentPosition(
                (position) => {
                    const lat = position.coords.latitude;
                    const lng = position.coords.longitude;
                    const geocoder = new google.maps.Geocoder();
                    const latlng = { lat: parseFloat(lat), lng: parseFloat(lng) };
                    geocoder.geocode({ location: latlng }, (results, status) => {
                        showLoader(false);
                        $('#useCurrentLocation').removeClass('loading').prop('disabled', false);
                        if (status === 'OK' && results[0]) {
                            $('#current_address').val(results[0].formatted_address);
                            showNotification('Current location fetched successfully', 'success');
                        } else {
                            showNotification('Unable to fetch location', 'error');
                        }
                    });
                },
                (error) => {
                    showLoader(false);
                    $('#useCurrentLocation').removeClass('loading').prop('disabled', false);
                    showNotification('Error fetching location: ' + error.message, 'error');
                }
            );
        });

        $('#multiStepForm').on('submit', function(e) {
            e.preventDefault();
            let valid = true;
            const role = selectedRole;
            if (!role) {
                showNotification('Please select a role', 'error');
                return;
            }
            if (role === 'doctor') {
                if (!$('#doctor_name').val()) valid = false;
                if (!$('#doctor_email').val()) valid = false;
                if (!$('#doctor_password').val()) valid = false;
                if ($('#doctor_password').val() !== $('#doctor_password_confirmation').val()) valid = false;
            } else if (role === 'patient') {
                if (!$('#patient_name').val()) valid = false;
                if (!$('#patient_email').val()) valid = false;
                if (!$('#patient_password').val()) valid = false;
                if ($('#patient_password').val() !== $('#patient_password_confirmation').val()) valid = false;
            }
            if (!valid) {
                showNotification('Please fill all required fields', 'error');
                return;
            }
            if (role === 'doctor') {
                $('#patient_name').prop('disabled', true);
                $('#gender').prop('disabled', true);
                $('#patient_email').prop('disabled', true);
                $('#patient_password').prop('disabled', true);
                $('#patient_password_confirmation').prop('disabled', true);
                var submitBtn = $('#doctorSubmitBtn');
            } else if (role === 'patient') {
                $('#doctor_name').prop('disabled', true);
                $('select[name="degrees[]"]').prop('disabled', true);
                $('#clinic_name').prop('disabled', true);
                $('select[name="specializations[]"]').prop('disabled', true);
                $('#country').prop('disabled', true);
                $('#state').prop('disabled', true);
                $('#city').prop('disabled', true);
                $('#address').prop('disabled', true);
                $('#current_address').prop('disabled', true);
                $('#doctor_email').prop('disabled', true);
                $('#doctor_password').prop('disabled', true);
                $('#doctor_password_confirmation').prop('disabled', true);
                var submitBtn = $('#patientSubmitBtn');
            }
            submitBtn.addClass('loading').prop('disabled', true);
            showLoader(true);
            $.ajax({
                url: '{{ route("register.submit") }}',
                type: 'POST',
                data: $(this).serialize(),
                success: function(response) {
                    showLoader(false);
                    submitBtn.removeClass('loading').prop('disabled', false);
                    if (response.success) {
                        showNotification(response.message, 'success');
                        setTimeout(() => window.location.href = response.redirect, 2000);
                    } else {
                        showNotification(response.message, 'error');
                    }
                },
                error: function(xhr) {
                    showLoader(false);
                    submitBtn.removeClass('loading').prop('disabled', false);
                    let errors = xhr.responseJSON.errors || {};
                    let errorMsg = Object.values(errors).flat().join('<br>');
                    showNotification(errorMsg || 'Registration failed', 'error');
                }
            });
        });
